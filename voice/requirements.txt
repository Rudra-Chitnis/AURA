# AURA Voice Client — Python Dependencies
# Install with: pip install -r requirements.txt
#
# Python 3.10–3.12 required.
# Python 3.13+ is NOT supported — ctranslate2 (faster-whisper dependency)
# does not yet publish binary wheels for 3.13.
# Install Python 3.11 or 3.12 from https://python.org/downloads/
#
# Windows: install a PortAudio-compatible audio backend for sounddevice.
# If "No module named '_sounddevice'" → reinstall: pip install sounddevice --force-reinstall
# If audio device errors occur → install Microsoft Visual C++ Redistributable.

# ── Speech-to-text ────────────────────────────
# ctranslate2 (transitive dep of faster-whisper) requires pre-built C wheels.
# Downloads Whisper model weights to ~/.cache/huggingface/ on first run.
faster-whisper>=1.0.0

# ── Text-to-speech ────────────────────────────
# edge-tts requires an active internet connection.
# Falls back to Windows SAPI (pyttsx3-free, via PowerShell) if edge-tts fails.
edge-tts>=6.1.0

# ── Audio I/O ─────────────────────────────────
sounddevice>=0.4.6
scipy>=1.11.0

# ── Numerical ─────────────────────────────────
numpy>=1.24.0,<2.0.0

# ── HTTP / backend comms ──────────────────────
requests>=2.31.0
